cmake_minimum_required(VERSION 3.20)
project(KompendiumPraktyka LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
option(KOMPENDIUM_SANITIZERS "Enable selected compiler sanitizers" OFF)
set(KOMPENDIUM_SANITIZER_SET "address,undefined" CACHE STRING "Compiler sanitizer list")

find_package(Threads REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()

add_library(kompendium INTERFACE)
target_include_directories(kompendium INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(kompendium INTERFACE cxx_std_23)
target_link_libraries(kompendium INTERFACE Threads::Threads)
if(MSVC)
    target_compile_options(kompendium INTERFACE /W4 /permissive- /utf-8)
else()
    target_compile_options(kompendium INTERFACE -Wall -Wextra -Wpedantic)
endif()
if(KOMPENDIUM_SANITIZERS)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(kompendium INTERFACE -fsanitize=${KOMPENDIUM_SANITIZER_SET} -fno-omit-frame-pointer)
        target_link_options(kompendium INTERFACE -fsanitize=${KOMPENDIUM_SANITIZER_SET})
    else()
        message(FATAL_ERROR "This sanitizer configuration requires Clang or GCC")
    endif()
endif()

foreach(name IN ITEMS parser money queue)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE kompendium)
    add_test(NAME ${name} COMMAND ${name}_test)
    set_tests_properties(${name} PROPERTIES TIMEOUT 15)
endforeach()
add_executable(stats src/stats.cpp)
target_link_libraries(stats PRIVATE kompendium)
add_test(NAME cli COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py" "$<TARGET_FILE:stats>")
set_tests_properties(cli PROPERTIES TIMEOUT 15)
